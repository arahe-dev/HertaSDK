// uniformKeyed: 256 yielding clients, uniform keys, Reject, 1 or 2 budgets.
func uniformKeyed(do func(context.Context, int) (int, error), over func(error) bool, keys int, window time.Duration) float64 {
	var stop atomic.Bool
	var wg sync.WaitGroup
	var ok atomic.Uint64
	for c := 0; c < 256; c++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			rng := rand.New(rand.NewPCG(uint64(c), 3))
			for !stop.Load() {
				if _, err := do(context.Background(), rng.IntN(keys)); err == nil {
					ok.Add(1)
				} else if over(err) {
					runtime.Gosched()
				}
			}
		}()
	}
	time.Sleep(window)
	stop.Store(true)
	wg.Wait()
	return float64(ok.Load()) / window.Seconds()
}

func TestPeek(t *testing.T) {
	w := 600 * time.Millisecond
	key := func(i int) string { return strconv.Itoa(i) }
	for _, keys := range []int{0, 64, 4096} {
		for _, res := range []int{1, 2} {
			var a, b []float64
			for r := 0; r < 3; r++ {
				{
					specs := []vk.ResourceSpec{{Name: "a", Capacity: 4}, {Name: "b", Capacity: 4}}[:res]
					rt, _ := vk.NewRuntime(specs...)
					reqs := []vk.Requirement{{Name: "a", Units: 1}, {Name: "b", Units: 1}}[:res]
					p := vk.Policy[int]{Effect: vk.Idempotent, Admission: vk.Reject, Resources: reqs}
					if keys > 0 {
						p.SerializeKey = key
					}
					op, _ := vk.NewOperation(rt, "op", func(_ context.Context, i int) (int, error) { return i, nil }, p)
					a = append(a, uniformKeyed(op.Do, func(e error) bool { return errors.Is(e, vk.ErrOverloaded) }, max(keys, 1), w))
				}
				{
					specs := []vp.ResourceSpec{{Name: "a", Capacity: 4}, {Name: "b", Capacity: 4}}[:res]
					rt, _ := vp.NewRuntime(specs...)
					reqs := []vp.Requirement{{Name: "a", Units: 1}, {Name: "b", Units: 1}}[:res]
					p := vp.Policy[int]{Effect: vp.Idempotent, Admission: vp.Reject, Resources: reqs}
					if keys > 0 {
						p.SerializeKey = key
					}
					op, _ := vp.NewOperation(rt, "op", func(_ context.Context, i int) (int, error) { return i, nil }, p)
					b = append(b, uniformKeyed(op.Do, func(e error) bool { return errors.Is(e, vp.ErrOverloaded) }, max(keys, 1), w))
				}
				runtime.GC()
			}
			fmt.Printf("keys=%-5d budgets=%d | with peek=%7.0fk  without=%7.0fk  ratio=%.2f\n", keys, res, median(a)/1e3, median(b)/1e3, median(a)/median(b))
		}
	}
}
