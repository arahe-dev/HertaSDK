package probe

import (
	"context"
	"errors"
	"os"
	"strings"

	orig "github.com/arahe-dev/hertasdk"
	v0 "github.com/arahe-dev/hertasdk/recon"
	va "probe/va"
	vb "probe/vb"
	vc "probe/vc"
)

type doer func(context.Context, int) (int, error)

type impl struct {
	name string
	mk   func(c int64, hold <-chan struct{}) (doer, func(error) bool, func())
}

var impls = []impl{
	{"orig", mkorig},
	{"recon", mkv0},
	{"A:fast-reject", mkva},
	{"B:A+padding", mkvb},
	{"C:full-flag", mkvc},
}

func must(err error) {
	if err != nil {
		panic(err)
	}
}

func mkorig(c int64, hold <-chan struct{}) (doer, func(error) bool, func()) {
	rt, _ := orig.NewRuntime(orig.ResourceSpec{Name: "r", Capacity: c})
	pol := orig.Policy[int]{Effect: orig.Idempotent, Admission: orig.Reject, Resources: []orig.Requirement{{Name: "r", Units: 1}}}
	op, err := orig.NewOperation(rt, "op", func(_ context.Context, i int) (int, error) { return i, nil }, pol)
	must(err)
	wpol := pol
	wpol.Admission = orig.Wait
	blk, err := orig.NewOperation(rt, "blk", func(_ context.Context, i int) (int, error) { <-hold; return i, nil }, wpol)
	must(err)
	park := func() { go blk.Do(context.Background(), 0) }
	return op.Do, func(e error) bool { return errors.Is(e, orig.ErrOverloaded) }, park
}

// pick returns the implementations named in $IMPLS (comma list), or all.
func pick() []impl {
	want := os.Getenv("IMPLS")
	if want == "" {
		return impls
	}
	var out []impl
	for _, im := range impls {
		if strings.Contains(","+want+",", ","+im.name+",") {
			out = append(out, im)
		}
	}
	return out
}
